Run All Test Suites

# Run all integration + e2e + regression tests
uv run pytest tests/integration/ tests/e2e/ tests/regression/ -v

# Same, but with minimal output (just pass/fail counts)
uv run pytest tests/integration/ tests/e2e/ tests/regression/ -q
Run Individual Test Suites

# Integration tests only
uv run pytest tests/integration/ -v

# E2E tests only
uv run pytest tests/e2e/ -v

# Regression tests only
uv run pytest tests/regression/ -v
Run Specific Test Files

# VPR pipeline contract tests
uv run pytest tests/integration/test_vpr_pipeline_contract.py -v

# Handler-DAL contract tests  
uv run pytest tests/integration/test_vpr_dal_handler_contract.py -v

# Schema serialization tests
uv run pytest tests/integration/test_vpr_schema_serialization.py -v

# FVS pipeline wiring tests
uv run pytest tests/integration/test_fvs_pipeline_wiring.py -v
Useful Test Options

# Stop on first failure (helpful for debugging)
uv run pytest tests/integration/ tests/e2e/ tests/regression/ -x

# Show print statements and logging output
uv run pytest tests/integration/ -v -s

# Show slowest 10 tests
uv run pytest tests/integration/ -v --durations=10

# Run only tests matching a pattern
uv run pytest tests/integration/ -v -k "camel_case"

# Run with coverage report
uv run pytest tests/integration/ tests/e2e/ tests/regression/ --cov=careervp --cov-report=html
