def extract_company_keywords(company_research: dict) -> list:
    """
    Extract ATS-relevant keywords from company research
    """
    keywords = []
    keywords.extend(company_research.get('products', []))
    keywords.extend(company_research.get('technologies', []))
    keywords.extend(company_research.get('industry_terms', []))
    return list(set(keywords))[:15]  # Top 15 unique keywords
